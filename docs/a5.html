Assignment 5!

For this assignment, I built a circuit that uses an IR remote to control the speed of a DC motor. I used an IR receiver to detect button presses and an N-MOSFET to control the motor with PWM. The plus and minus buttons adjust the motor speed, while the 0 button stops it. I also used a flyback diode to protect the circuit and an external power supply to power the motor.

Ardunio code snippet:

       // Include the IR remote library
        #include IRremote.hpp
        // IR receiver signal connects to pin 7
        const int IR_PIN = 7;  
        // MOSFET gate control will use PWM pin 9
        const int MOTOR_PIN = 9;
        //  intializing a variable to store the motor speed 
        int motorSpeed = 0;

        void setup() {
            // Start Serial Monitor to display remote button codes
            Serial.begin(9600);
            //Sets the motor control pin 9 as an output, it starts LOW state on Arduino
            pinMode (MOTOR_PIN, OUTPUT);
            // Makes the motor off initailly 
            analogWrite(MOTOR_PIN, 0);
            // Initialize the IR receiver on pin 7
            IrReceiver.begin(IR_PIN, ENABLE_LED_FEEDBACK);

        }

        void loop() {
            // Check whether the receiver detected an IR signal
            if (IrReceiver.decode()) {
                // Check if the received signal is not a repeated or invalid command
                if (!(IrReceiver.decodedIRData.flags & IRDATA_FLAGS_IS_REPEAT)
                && IrReceiver.decodedIRData.protocol != UNKNOWN) {

                // intialize a varible to store readings from the remote
                int command = IrReceiver.decodedIRData.command;
                
                // Only process the three remote buttons
                if (command == 70 || command == 21 || command == 22) {

                // Print the button's command code
                    Serial.println(command);
                    // pluse button: 70 means higher speed
                    if (command == 70){
                    motorSpeed += 50;
                    }
                    //minus button: 21 means lower speed
                    else if (command == 21){
                    motorSpeed -= 50;
                    }
                    //stop motor if 0 is pressed: 22 means stop motor
                    else if (command == 22) {
                    motorSpeed = 0;
                    }
                    // Prevent values below 0 or above 255
                    motorSpeed = constrain(motorSpeed, 0, 255);

                    // Send the updated PWM value to pin 9
                    analogWrite(MOTOR_PIN, motorSpeed);

                    

                    // Print the current motor PWM value
                    Serial.println(motorSpeed);
                }
                }
                // Prepare the receiver for the next signal
                IrReceiver.resume();

            }
        }

            int enable1 = 7; // Pin 7 connected to enable1 
            int enable2 = 8; // Pin 8 connected to enable2
        
            int motor_ccw1 = 6; // Pin 6 connected to motor control 1A 
            int motor_ccw2 = 10; // Pin 10 connected to 3A
                
            int motor_cw1 = 3; //Pin 3 connected to 2A
            int motor_cw2 = 13;  // Pin 13 connected to motor control 4A
            

            void setup() {
            // Set the pins
            pinMode(enable1, OUTPUT);
            pinMode(motor_ccw1, OUTPUT);
            pinMode(motor_cw1, OUTPUT);

            pinMode(enable2, OUTPUT);
            pinMode(motor_ccw2, OUTPUT);
            pinMode(motor_cw2, OUTPUT);

            // Enable the first motor driver all the time
            digitalWrite(enable1, HIGH);
            digitalWrite(enable2, HIGH);
            }
            void loop() {
            //brake the motor
            digitalWrite(motor_ccw1, LOW);
            digitalWrite(motor_cw1, LOW);
            digitalWrite(motor_ccw2, LOW);
            digitalWrite(motor_cw2, LOW);
            delay(100);
            //run both motor CCW 
            digitalWrite(motor_ccw1, HIGH);
            digitalWrite(motor_cw1, LOW);
            digitalWrite(motor_ccw2, HIGH);
            digitalWrite(motor_cw2, LOW);
            delay(1000);
            //stop both motors
            digitalWrite(motor_ccw1, LOW);
            digitalWrite(motor_cw1, LOW);
            digitalWrite(motor_ccw2, LOW);
            digitalWrite(motor_cw2, LOW);
            delay(100);
            //run both motor CW
            digitalWrite(motor_ccw1, LOW);
            digitalWrite(motor_cw1, HIGH);
            digitalWrite(motor_ccw2, LOW);
            digitalWrite(motor_cw2, HIGH);
            delay(1000);

            //run one motor CCW, one CW
            digitalWrite(motor_ccw1, HIGH);
            digitalWrite(motor_cw1, LOW);

            delay(1000);

            }

4: I used AI to help me check my circuit connections since I was not familiar with the power supply module. I also used it to debug my readings and code. One problem I faced was that the IR receiver stopped responding to button presses when the motor speed increased, and the Serial Monitor kept showing a command value of 0. AI helped me understand that electrical noise from the motor could be interfering with the IR receiver. To improve this, I added two ceramic capacitors, one across the motor and another across the IR receiver's power connections. I also connected the IR receiver's ground directly to the Arduino GND and updated my code to ignore repeated and invalid commands. These changes improved the remote's response, although some lag remained at higher motor speeds. I also discussed these issues with the instructor who recommened adding the 2 capacitors to reduce the noise .
